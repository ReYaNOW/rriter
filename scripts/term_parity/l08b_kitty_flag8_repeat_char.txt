# l08b: the rest of L8 (Kitty flag 8, repeat, `key --char`). l08_kitty_keys.txt covers flags 0/1/3,
# Ctrl+C, Shift+Enter, Escape, release and IME; this file does not repeat them.
# Reader: function `r` as in l08_kitty_keys.txt (raw mode, READY, hex of the bytes until 1 s of silence, END).
# Run: python3 scripts/term_parity/compare.py scripts/term_parity/l08b_kitty_flag8_repeat_char.txt
#
# Expected bytes (from parity spec 5.3 and the Kitty protocol; not from either app's output):
#   flags 8, unmodified keys (no modifier-key reports: their codes 57441-57444 are not in the spec table):
#     a press            CSI 97 u          = 1b5b393775
#     Enter press        CSI 13 u          = 1b5b313375   (Enter is not legacy at flag 8)
#     Escape press       CSI 27 u          = 1b5b323775
#     `key --repeat 3 a` 4 x CSI 97 u      = 1b5b3937751b5b3937751b5b3937751b5b393775
#       (no flag 2: repeats are reported as presses, no release)
#   flags 0, `key --char с ctrl+c` (Cyrillic с, Ctrl by physical key): 03
#   flags 1, `key --char с ctrl+c`: CSI 99;5u = 1b5b39393b3575   (spec decision 18)
#   flags 1, `key --repeat 2 a`: text path, a x3 = 616161        (repeats of text keys without flag 2)
#   flags 3 (1+2), `key --repeat 2 escape`: press CSI 27u = 1b5b323775; repeat CSI 27;1:2u = 1b5b32373b313a3275
#       x2; release CSI 27;1:3u = 1b5b32373b313a3375
#
# The reader is started with `type r\n` (text commit with LF) once flag 8 is set: at flag 8 Enter is
# not a legacy CR, so the shell would not run the command.
#@click_ui WelcomeIdeMode
idle 300
#@click_ui SidebarSlot(Terminal)
idle 3000
#@click_ui TerminalBody
idle 800
type cd /tmp
idle 300
key enter
idle 800
type r() { stty raw -echo min 1 time 0; printf 'RE''ADY\r\n'; { dd bs=1 count=1; stty min 0 time 10; dd bs=1; } 2>/dev/null | od -An -tx1 -v | tr -d ' \012'; printf '\r\n'; stty sane; }
idle 300
key enter
idle 800
# flags 8 (push at flags 0: Enter is still CR here, so `key enter` runs the command)
type printf '\033[>8u'
idle 300
key enter
idle 800
type r\n
idle 300
key a
idle 300
key enter
idle 300
key escape
idle 300
key --repeat 3 a
idle 1500
dump
# pop flag 8 (flags 8 is active: the command is submitted with LF)
type printf '\033[<u'\n
idle 800
# flags 0 with `key --char`: Ctrl+С by physical key, legacy byte 03
type r\n
idle 300
key --char с ctrl+c
idle 1500
dump
# flags 1: Ctrl+С in the Cyrillic layout -> CSI 99;5u; plain `a` repeats are text
type printf '\033[>1u'\n
idle 800
type r\n
idle 300
key --char с ctrl+c
idle 300
key --repeat 2 a
idle 1500
dump
# flags 3 (on top of 1): repeat of a Sent key with event types
type printf '\033[>3u'\n
idle 800
type r\n
idle 300
key --repeat 2 escape
idle 1500
dump
